class Solution {
  public:
    int kthSmallest(vector<int>& arr, int k) {
        int mx = *max_element(arr.begin(), arr.end());
        vector<int> freq(mx + 1, 0);
    
        for (int x : arr) freq[x]++;
    
        for (int i = 0; i <= mx; i++) {
            k -= freq[i];
            if (k <= 0) return i;
        }
        return -1;
    }
};
