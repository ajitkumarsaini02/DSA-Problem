class Solution {
public:
    char findTheDifference(string s, string t) {
        char str;
        int n = t.size();
        vector<int> freq(26, 0);
        for(int i=0;i<n;i++){
            freq[t[i]-'a']++;
        }
        for(int i=0;i<s.size();i++){
            freq[s[i]-'a']--;
        }
        for(int i=0;i<26;i++){
            if(freq[i] == 1){
                str = i + 'a';
            }
        }
        return str;
    }
};
