class Solution {
public:
    vector<int>dat;
    int climb(int n){
        if(n==0) return 1;
        else if(n<0) return 0;
        else{
            if(dat[n]==0){
                dat[n]=climb(n-1)+climb(n-2);
            }
            return dat[n];
        }   
    }
    int climbStairs(int n) {
        dat.resize(n+1,0);
        return climb(n);
    }
};



class Solution {
public:
    vector<int>dat;
    int climb(int n){
        if(n==0) return 1;
        if(n<0) return 0;
        if (dat[n] != 0) return dat[n];
        
        dat[n]=climb(n-1)+climb(n-2);
        return dat[n];
    }
    int climbStairs(int n) {
        dat.resize(n+1,0);
        return climb(n);
    }
};



class Solution {
public:
    int climbStairs(int n) {
        if(n<=1) return 1;
        int a=1,b=1;
        for(int i=2;i<=n;i++){
            int temp=a+b;
            a=b;
            b=temp;
        }
        return b;
    }
};
