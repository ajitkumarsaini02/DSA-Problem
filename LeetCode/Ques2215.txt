class Solution {
public:
    vector<vector<int>> findDifference(vector<int>& nums1, vector<int>& nums2) {
        vector<int> freq1(2001, 0), freq2(2001, 0);
        for(int n : nums1){
            freq1[n + 1000] = 1;
        }
        for(int n : nums2){
            freq2[n + 1000] = 1;
        }
        vector<int> a, b;
        for(int i=0;i<freq1.size();i++){
            if(freq1[i] == 1 && freq2[i] == 0){
                a.push_back(i - 1000);
            }
        }
        for(int i=0;i<freq2.size();i++){
            if(freq2[i] == 1 && freq1[i] == 0){
                b.push_back(i - 1000);
            }
        }
        return {a, b};
    }
};
