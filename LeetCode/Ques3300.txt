class Solution {
public:
    int sum(int n){
        int t = 0;
        while (n > 0){
            t += (n % 10);
            n /= 10;
        }
        return t;
    }

    int minElement(vector<int>& nums) {
        int n = nums.size();
        for(int i=0;i<n;i++){
            int t = sum(nums[i]);
            nums[i] = t;
        }
        return *min_element(nums.begin(), nums.end());
    }
};
