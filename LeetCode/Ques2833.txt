class Solution {
public:
    int furthestDistanceFromOrigin(string moves) {
        int n = moves.size();
        int cL = 0, cR = 0;
        for(auto s : moves){
            if(s == 'L') cL++;
            else if(s == 'R') cR++;
        }
        if(cL >= cR){
            return (n - 2 * cR);
        }
        else{
            return (n - 2 * cL);
        }
    }
};
