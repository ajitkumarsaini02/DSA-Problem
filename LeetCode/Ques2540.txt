class Solution {
public:
    int getCommon(vector<int>& nums1, vector<int>& nums2) {
        int i=0;
        int j=0;

        while(i < nums1.size() && j < nums2.size()){
            if(nums1[i] == nums2[j]){
                return nums1[i];
            }
            else if(nums1[i] < nums2[j]){
                i++;
            }
            else{
                j++;
            }
        }
        return -1;
    }
};


class Solution {
public:
    int getCommon(vector<int>& nums1, vector<int>& nums2) {
        int mn = INT_MAX;
        int i=0;
        int j=0;
        int n = nums1.size();
        int m = nums2.size();

        while(i < n && j < m){
            if(nums1[i] == nums2[j] && nums1[i] < mn){
                mn = min(nums1[i], mn);
                i++;
                j++;
            }
            else if(nums1[i] < nums2[j]){
                i++;
            }
            else{
                j++;
            }
        }
        if(mn == INT_MAX){
            return -1;
        }
        else{
            return mn;
        }
    }
};
