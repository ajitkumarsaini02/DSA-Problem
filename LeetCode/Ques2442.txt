class Solution {
public:
    int reverseNum(int n){
        int num = 0;
        while(n > 0){
            num = num * 10 + (n % 10);
            n /= 10;
        }
        return num;
    }

    int countDistinctIntegers(vector<int>& nums) {
        int n = nums.size();
        for(int i = 0; i < n; i++){
            nums.push_back(reverseNum(nums[i]));
        }
        sort(nums.begin(), nums.end());

        int ans = 0;
        for(int i = 0; i < nums.size(); i++){
            if(i == 0 || nums[i] != nums[i-1]){
                ans++;
            }
        }

        return ans;
    }
};

class Solution {
public:
    int reverse(int n){
        int rev = 0;
        while(n > 0){
            int d = n % 10;
            rev = rev * 10 + d;
            n /= 10;
        }
        return rev;
    }

    int countDistinctIntegers(vector<int>& nums) {
        int n = nums.size();
        unordered_set<int> mp;
        
        for(int i=0;i<n;i++){
            mp.insert(nums[i]);
            int d = reverse(nums[i]);
            mp.insert(d);
        }
        return mp.size();
    }
};
