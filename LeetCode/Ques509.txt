class Solution {
public:
    vector<int>dat;
    int myFib(int n){
        if(n<2) return n;
        else{
            dat[n]=myFib(n-1)+myFib(n-2);
            return dat[n];        
        } 
    }
    int fib(int n) {
        dat.resize(n+1);
        return myFib(n);
    }
};




class Solution {
public:
    int myFib(int n,vector<int>&dat){
        if(n<2) return n;
        else{
            if(dat[n]==0)
                dat[n]=myFib(n-1,dat)+myFib(n-2,dat);
            return dat[n];        
        } 
    }
    int fib(int n) {
        vector<int>dat(n+1,0);
        return myFib(n,dat);
    }
};




class Solution {
public:
    int fib(int n) {
        if(n==0) return 0;
        else if(n==1) return 1;
        else return fib(n-1)+fib(n-2);
    }
};




class Solution {
public:
    int fib(int n) {
        if(n==0) return 0;
        if(n==1) return 1;
        int a=0, b=1,c;
        for(int i=2;i<=n;i++){
            c=a+b;
            a=b;
            b=c;
        }
        return c;
    }
};
