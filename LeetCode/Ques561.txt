class Solution {
public:
    int arrayPairSum(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        
        int n = nums.size();
        int sum = 0;
        for(int i=0;i<n;i+=2){
            sum += nums[i];
        }
        return sum;
    }
};

class Solution {
public:
    int arrayPairSum(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        int n = nums.size();
        int mn = INT_MAX;
        int sum = 0;
        for(int i=0;i<n-1;i+=2){
            mn = min(nums[i], nums[i+1]);
            sum += mn;
        }
        return sum;
    }
};
