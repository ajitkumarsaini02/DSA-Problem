class Solution {
public:
    bool isPrime(int n) {
        if (n < 2) {
            return false;
        }
        if(n==2){
            return true;
        }
        if(n%2==0){
            return false;
        }
        for (int i = 3; i*i <= n; i++) {
            if (n % i == 0) {
                return false;
            }
        }
        return true;
    }
    bool checkPrimeFrequency(vector<int>& nums) {
        int n = nums.size();
        int mx = *max_element(nums.begin(), nums.end());
        vector<int> a(mx+1,0);
        for(int i=0;i<n;i++){
            a[nums[i]]++;
        }

        for(int i=0;i<a.size();i++){
            if(isPrime(a[i])){
                return true;
            }
        }
        return false;
    }
};
