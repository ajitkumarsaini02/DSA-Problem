class Solution {
public:
    bool check(vector<int>& nums) {
        int n=nums.size();
        bool rotated = false;
        for(int i=1;i<n;i++){
            if(nums[i] < nums[i-1]){
                if(rotated) return false;
                rotated = true;
            }
        }
        if(rotated && nums[0] < nums[n-1]) return false;
        return true;
    }
};
