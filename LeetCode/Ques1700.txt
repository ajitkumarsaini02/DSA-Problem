class Solution {
public:
    int countStudents(vector<int>& st, vector<int>& sd) {
        int count = 0;
        while(!st.empty() && count < st.size()){
            if(st[0] == sd[0]){
                st.erase(st.begin());
                sd.erase(sd.begin());
                count = 0;
            }
            else{
                st.push_back(st[0]);
                st.erase(st.begin());
                count++;
            }
        }
        return st.size();
    }
};
