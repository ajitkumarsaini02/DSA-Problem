class Solution {
public:
    bool isPrime(int n) {
        if (n < 2) {
            return false;
        }
        if(n==2){
            return true;
        }
        if(n%2==0){
            return false;
        }
        for (int i = 3; i*i <= n; i++) {
            if (n % i == 0) {
                return false;
            }
        }
        return true;
    }
    int maximumPrimeDifference(vector<int>& nums) {
        vector<int> idx;
        for(int i=0;i<nums.size();i++){
            if(isPrime(nums[i])){
                idx.push_back(i);
            }
        }
        int mx = *max_element(idx.begin(), idx.end());
        int mn = *min_element(idx.begin(), idx.end());
        if(nums.size()==1){
            return 0;
        }
        else{
            return abs(mx-mn);
        }
    }
};
