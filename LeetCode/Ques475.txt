class Solution {
public:
    int findRadius(vector<int>& houses, vector<int>& heat) {
        int n = houses.size();
        int m = heat.size();
        int mx = 0;
        sort(houses.begin(), houses.end());
        sort(heat.begin(), heat.end());
        for(int i=0;i<n;i++){
            int mn = INT_MAX;
            for(int j=0;j<m;j++){
                mn = min(mn, abs(houses[i] - heat[j]));
            }
            mx = max(mx, mn);
        }
        return mx;
    }
};
