class Solution {
public:
    bool bfs(int i, int j, vector<vector<char>>& grid,vector<vector<int>>& vis){
        int n = grid.size();
        int m = grid[0].size();

        queue<pair<pair<int,int>, pair<int,int>>>q;

        q.push({{i, j},{-1, -1}});
        vis[i][j] = 1;
        char ch = grid[i][j];

        int dx[] = {0, 0, -1, 1};
        int dy[] = {1, -1, 0, 0};

        while(!q.empty()){
            auto curr = q.front();
            q.pop();

            int x = curr.first.first;
            int y = curr.first.second;
            int px = curr.second.first;
            int py = curr.second.second;
            for(int d=0;d<4;d++){
                int nx = x + dx[d];
                int ny = y + dy[d];
                
                if(nx >= 0 && ny >= 0 && nx < n && ny < m &&grid[nx][ny] == ch){
                    if(!vis[nx][ny]){
                        vis[nx][ny] = 1;
                        q.push({{nx,ny}, {x, y}});
                    }
                    else if(nx != px && ny != py){
                        return true;
                    }
                }
            }
        }
        return false;
    }
    bool containsCycle(vector<vector<char>>& grid) {
        int n = grid.size();
        int m = grid[0].size();

        vector<vector<int>> vis(n, vector<int>(m, 0));

        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(!vis[i][j]){
                    if(bfs(i, j, grid, vis)){
                        return true;
                    }
                }
            }
        }
        return false;
    }
};
