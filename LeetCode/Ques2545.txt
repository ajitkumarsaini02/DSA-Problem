class Solution {
public:
    vector<vector<int>> sortTheStudents(vector<vector<int>>& s, int k) {
        int n = s.size();
        for(int i=0;i<n-1;i++){
            for(int j=0;j<n-i-1;j++){
                if(s[j][k] < s[j+1][k]){
                    swap(s[j], s[j+1]);
                }
            }
        }
        return s;
    }
};
