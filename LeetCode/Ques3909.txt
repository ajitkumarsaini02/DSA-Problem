class Solution {
public:
    int compareBitonicSums(vector<int>& nums) {
        int n = nums.size();
        int idx = -1;

        for(int i=1;i<n-1;i++){
            if((nums[i] > nums[i-1] && nums[i] > nums[i+1])){
                idx = i;
                break;
            }
        }

        if(idx == -1){
            return -1;
        }

        long long sumInc = 0;
        long long sumDec = 0;

        for(int i=0;i<n;i++){
            if(i < idx){
                sumInc += nums[i];
            }
            else if(i > idx){
                sumDec += nums[i];
            }
        }

        if(sumInc > sumDec) return 0;
        if(sumInc < sumDec) return 1;
        return -1;
    }
};
