class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        int mx = *max_element(nums.begin(), nums.end());
        int mn = *min_element(nums.begin(), nums.end());
        int n = nums.size();
        int size = mx - mn +1;
        vector<int> freq(size, 0);
        for(int i=0;i<n;i++){
            freq[nums[i] - mn]++;
        }
        vector<int> ans;
        for(int j=0;j<k;j++){
            int idx = -1;
            int best = 0;
            for(int i=0;i<size;i++){
                if(best < freq[i]){
                    idx = i;
                    best = freq[i];
                }
            }
            ans.push_back(idx + mn);
            freq[idx] = 0;
        }
        
        
        return ans;
    }
};
