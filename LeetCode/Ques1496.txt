class Solution {
public:
    bool isPathCrossing(string path) {
        vector<pair<int, int>> pos;
        int x = 0, y = 0;
        pos.push_back({x, y});
        for(char ch : path) {
            if(ch == 'N') y++;
            else if(ch == 'S') y--;
            else if(ch == 'E') x++;
            else if(ch == 'W') x--;

            for(int i=0;i<pos.size();i++){
                if(pos[i].first == x && pos[i].second == y){
                    return true;
                }
            }
            pos.push_back({x, y});

        }
        return false;
    }
};
