class Solution {
public:
    vector<int> findGoodIntegers(int n) {
        vector<int> ans;
        int limit = cbrt(n);
        unordered_map<long long, int> mp;
        for(int a=1;a<=limit;a++){
            for(int b=a;b<=limit;b++){
                long long sum = 1LL*a*a*a + 1LL*b*b*b;
                if(sum > n) break;
                mp[sum]++;
            }
        }

        for(auto it : mp){
            if(it.second >= 2){
                ans.push_back(it.first);
            }
        }
        sort(ans.begin(), ans.end());
        return ans;
    }
};
