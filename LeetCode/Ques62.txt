class Solution {
public:
    int dat[101][101]={0};
    int myfun(int i,int j,int m,int n){
        if(i==m-1 && j==n-1)
            return 1;
        if(i==m || j==n)
            return 0;

        else{
            if(dat[i][j]==0){
                dat[i][j] = myfun(i + 1, j, m, n) + myfun(i, j + 1, m, n);
                
            }
            return dat[i][j];
                    
        }    
    }
    int uniquePaths(int m, int n) {
       return myfun(0,0,m,n); 
    }
};
